const downloadFile = (id, imageElementId) => {
  const response = await fetch('@api.routes.MediaDSController.getItemApi(dataset.getId(), "FILENAME").absoluteURL(request, request.secure())', {
    headers: {
      'api_token': '@dataset.configuration(Dataset.API_TOKEN, "TOKEN")',
      'participant_id': id
    }
  });

  if (!response.ok) {
    throw new Error(`HTTP error! status: ${response.status}`);
  }

  const imageBlob = await response.blob();
  const imageObjectURL = URL.createObjectURL(imageBlob);

  const imgElement = document.getElementById(imgElementId);
  imgElement.src = imageObjectURL;
};

// use it
const participantId = 'PARTICIPANT_ID' // looks like: u1234567890abcdef

    Download an image from the dataset 
    import requests
import shutil

url = '@api.routes.MediaDSController.getItemApi(dataset.getId(), "FILENAME").absoluteURL(request, request.secure())'
headers = {
    'api_token': '@dataset.configuration(Dataset.API_TOKEN, "TOKEN")',
    'participant_id': 'PARTICIPANT_ID'
}

response = requests.get(url, headers=headers, stream=True)
if response.status_code == 200:
    with open('FILENAME', 'wb') as f:
        response.raw.decode_content = True
        shutil.copyfileobj(response.raw, f)
else:
    print(f"Failed to download. Status code: {response.status_code}")
    Upload an image to the dataset 
    import requests

url = '@api.routes.MediaDSController.addItemApi(dataset.getId()).absoluteURL(request, request.secure())'
headers = {
    'api_token': '@dataset.configuration(Dataset.API_TOKEN, "TOKEN")',
    'participant_id': 'PARTICIPANT_ID'
}
files = {'files': open('/path/to/your/image.jpg', 'rb')}

response = requests.post(url, headers=headers, files=files)
if response.status_code == 200:
    print("Upload successful!")
else:
    print(f"Failed to upload. Status code: {response.status_code}")
