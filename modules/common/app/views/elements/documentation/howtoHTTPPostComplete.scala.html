const uploadFile = async () => {
  const myform = document.querySelector('#myform');
  const formData = new FormData(myform);

  const response = await fetch('@api.routes.CompleteDSController.addItemApi(dataset.getId()).absoluteURL(request, request.secure())', {
    method: 'POST',
    headers: {
      'api_token': '@dataset.configuration(Dataset.API_TOKEN, "TOKEN")'
    },
    body: formData
  });

  if (!response.ok) {
    throw new Error(`HTTP error! status: ${response.status}`);
  }
  
  const result = await response.json();
  console.log("Uploaded successfully:", result);
};

  • Python
    Upload a file to the dataset 
    import requests

url = '@api.routes.CompleteDSController.addItemApi(dataset.getId()).absoluteURL(request, request.secure())'
headers = {
    'api_token': '@dataset.configuration(Dataset.API_TOKEN, "TOKEN")'
}
files = {'files': open('/path/to/your/file.json', 'rb')}
data = {'description': 'My file upload'}

response = requests.post(url, headers=headers, files=files, data=data)
if response.status_code == 200:
    print("Upload successful!", response.json())
else:
    print(f"Failed to upload. Status code: {response.status_code}")
    Download a file from the dataset 
    import requests
import shutil

url = '@api.routes.CompleteDSController.downloadLatestFile(dataset.getId(), "FILENAME").absoluteURL(request, request.secure())'
# Must provide session/cookies if authentication is required
session_cookies = {'PLAY_SESSION': 'YOUR_SESSION_VALUE'}

response = requests.get(url, cookies=session_cookies, stream=True)
if response.status_code == 200:
    with open('FILENAME', 'wb') as f:
        response.raw.decode_content = True
        shutil.copyfileobj(response.raw, f)
else:
    print(f"Failed to download. Status code: {response.status_code}")
